//! 字节码编译器模块 (Bytecode Compiler) — 简化明文版
//! 负责将 VM IR 编译为明文字节码流

use crate::vm::arch;
use crate::vm::ir::VmInstruction;
use crate::vm::opcode::VmOpcode;
use std::collections::HashMap;

/// 待修补的跳转指令信息
struct PatchInfo {
    /// 字节码中操作数的起始偏移量
    operand_offset: usize,
    /// 跳转目标的标签 ID
    target_label: u32,
}

/// 原生孤岛的修补信息
pub struct IslandPatch {
    /// 在 bytecode 中的偏移，占 8 个字节
    pub bytecode_offset: usize,
    /// 对应的原生代码字节
    pub native_bytes: Vec<u8>,
}

pub struct BytecodeCompiler;

impl BytecodeCompiler {
    pub fn new() -> Self {
        Self
    }

    /// 将给定的 VM IR 序列编译为明文 Vec<u8>
    /// 返回 (bytecode, islands_to_patch)
    pub fn compile_block(&self, vm_irs: &[VmInstruction]) -> (Vec<u8>, Vec<IslandPatch>) {
        let mut bytecode = Vec::new();
        let mut label_positions: HashMap<u32, usize> = HashMap::new();
        let mut patches: Vec<PatchInfo> = Vec::new();
        let mut islands = Vec::new();

        // 模拟 VPushImm64 + VExec 的 opcode byte (使用 arch 中的映射，不过此处简单写死或查找)
        // 我们需要知道 VPushImm64 和 VExec 的 opcode byte
        let push_imm64_op = arch::opcode_to_byte(&VmOpcode::VPushImm64(0));
        let exec_op = arch::opcode_to_byte(&VmOpcode::VExec);

        for ir in vm_irs {
            // 如果指令包含原生孤岛字节，生成 VPushImm64(占位) + VExec
            if let Some(native_bytes) = &ir.native_bytes {
                bytecode.push(push_imm64_op);
                let patch_offset = bytecode.len();
                bytecode.extend_from_slice(&[0u8; 8]); // 8 字节占位符

                bytecode.push(exec_op);

                islands.push(IslandPatch {
                    bytecode_offset: patch_offset,
                    native_bytes: native_bytes.clone(),
                });
                continue;
            }

            match &ir.opcode {
                VmOpcode::VLabel(id) => {
                    label_positions.insert(*id, bytecode.len());
                    continue;
                }
                VmOpcode::VJmp(target_label) => {
                    let op_byte = arch::opcode_to_byte(&ir.opcode);
                    bytecode.push(op_byte);

                    patches.push(PatchInfo {
                        operand_offset: bytecode.len(),
                        target_label: *target_label,
                    });
                    bytecode.extend_from_slice(&[0u8; 4]);
                }
                VmOpcode::VJcc(_cond, target_label) => {
                    let op_byte = arch::opcode_to_byte(&ir.opcode);
                    bytecode.push(op_byte);

                    patches.push(PatchInfo {
                        operand_offset: bytecode.len(),
                        target_label: *target_label,
                    });
                    bytecode.extend_from_slice(&[0u8; 4]);
                }
                VmOpcode::VPushReg(offset) => {
                    bytecode.push(arch::opcode_to_byte(&ir.opcode));
                    bytecode.push(*offset as u8);
                }
                VmOpcode::VPopReg(offset) => {
                    bytecode.push(arch::opcode_to_byte(&ir.opcode));
                    bytecode.push(*offset as u8);
                }
                VmOpcode::VPushCtxReg(offset) => {
                    bytecode.push(arch::opcode_to_byte(&ir.opcode));
                    bytecode.push(*offset as u8);
                }
                VmOpcode::VPushImm32(val) => {
                    bytecode.push(arch::opcode_to_byte(&ir.opcode));
                    bytecode.extend_from_slice(&val.to_le_bytes());
                }
                VmOpcode::VPushImm64(val) => {
                    bytecode.push(arch::opcode_to_byte(&ir.opcode));
                    bytecode.extend_from_slice(&val.to_le_bytes());
                }
                _ => {
                    bytecode.push(arch::opcode_to_byte(&ir.opcode));
                }
            }
        }

        // 修补内部标签跳转
        for patch in &patches {
            if let Some(&target_pos) = label_positions.get(&patch.target_label) {
                let current_pos = patch.operand_offset + 4;
                let offset = target_pos as i64 - current_pos as i64;
                bytecode[patch.operand_offset..patch.operand_offset + 4]
                    .copy_from_slice(&(offset as i32).to_le_bytes());
            }
        }

        (bytecode, islands)
    }
}

impl Default for BytecodeCompiler {
    fn default() -> Self {
        Self::new()
    }
}
