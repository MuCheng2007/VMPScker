//! 线索化分发器 (Threaded Dispatcher) — 简化明文版

use crate::vm::arch::ArchConfig;
use iced_x86::code_asm::*;
use iced_x86::IcedError;

pub struct DispatcherGen;

impl DispatcherGen {
    /// 追加"获取-跳转"逻辑：读取 1 字节明文 opcode，查表间接跳转
    pub fn append_dispatch_logic(
        asm: &mut CodeAssembler,
        arch: &ArchConfig,
    ) -> Result<(), IcedError> {
        let ctx = &arch.context;

        // 1. Fetch: movzx scratch1_32, byte ptr [VIP] (auto zero-extends to 64-bit)
        asm.movzx(ctx.scratch1_32, byte_ptr(ctx.vip))?;

        // 2. VIP += 1
        asm.add(ctx.vip, 1_i32)?;

        // 3. Indirect Jump: jmp qword ptr [table + scratch1 * 8]
        asm.jmp(qword_ptr(ctx.table + ctx.scratch1 * 8))?;

        Ok(())
    }
}
