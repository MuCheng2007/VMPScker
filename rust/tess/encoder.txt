//! 编码器模块
//!
//! 将 VMProtect IR 指令编码为 x86/x64 机器码。
//! 使用 iced-x86 库 provide 高性能的编码功能。

use super::{
    DisassemblyMode, DisassemblyResult, DisassemblyError,
    ir::{IrInstruction, IrOpcode, IrOperand, IrRegister, IrJumpTarget},
};
use iced_x86::{Instruction as IcedInstruction, Register as IcedRegister, Code};

pub struct Encoder {
    mode: DisassemblyMode,
}

impl Encoder {
    pub fn new(mode: DisassemblyMode) -> Self {
        Self { mode }
    }

    pub fn encode(&self, instruction: &IrInstruction, base_address: u64) -> DisassemblyResult<EncodedInstruction> {
        let mut iced_insn = match &instruction.opcode {
            IrOpcode::Nop => IcedInstruction::with(Code::Nopd),
            IrOpcode::Mov { dst, src } => {
                let mut insn = IcedInstruction::with(Code::Mov_rm64_r64);
                self.set_operand(&mut insn, 0, dst)?;
                self.set_operand(&mut insn, 1, src)?;
                insn
            }
            IrOpcode::Push { src } => {
                let mut insn = IcedInstruction::with(Code::Push_r64);
                self.set_operand(&mut insn, 0, src)?;
                insn
            }
            IrOpcode::Pop { dst } => {
                let mut insn = IcedInstruction::with(Code::Pop_r64);
                self.set_operand(&mut insn, 0, dst)?;
                insn
            }
            IrOpcode::Jmp { target } => {
                match target {
                    IrJumpTarget::Direct(addr) => IcedInstruction::with_branch(Code::Jmp_rel8_64, *addr)
                        .map_err(|e| DisassemblyError::DecodeError(format!("Branch encoding error: {}", e)))?,
                    IrJumpTarget::Relative(off) => IcedInstruction::with_branch(Code::Jmp_rel8_64, instruction.rva.wrapping_add(*off as u64))
                        .map_err(|e| DisassemblyError::DecodeError(format!("Branch encoding error: {}", e)))?,
                    _ => return Err(DisassemblyError::UnsupportedInstruction("Unsupported jump target".to_string())),
                }
            }
            _ => return Err(DisassemblyError::UnsupportedInstruction(format!("Unsupported opcode for encoding: {:?}", instruction.opcode))),
        };

        iced_insn.set_next_ip(base_address.wrapping_add(instruction.rva));

        let mut encoder = iced_x86::Encoder::new(self.bitness());
        let len = encoder.encode(&iced_insn, base_address).map_err(|e| DisassemblyError::DecodeError(format!("Iced encoding error: {}", e)))?;
        
        Ok(EncodedInstruction {
            bytes: encoder.take_buffer(),
            len: len as u8,
        })
    }

    fn bitness(&self) -> u32 {
        match self.mode {
            DisassemblyMode::Mode16 => 16,
            DisassemblyMode::Mode32 => 32,
            DisassemblyMode::Mode64 => 64,
        }
    }

    fn set_operand(&self, insn: &mut IcedInstruction, op_index: u32, operand: &IrOperand) -> DisassemblyResult<()> {
        match operand {
            IrOperand::Register(reg) => {
                insn.set_op_register(op_index, self.convert_register(*reg));
            }
            IrOperand::Immediate(imm) => {
                // Set op kind first, then set immediate value
                match imm.size_bits() {
                    8 => insn.set_op_kind(op_index, iced_x86::OpKind::Immediate8),
                    16 => insn.set_op_kind(op_index, iced_x86::OpKind::Immediate16),
                    32 => insn.set_op_kind(op_index, iced_x86::OpKind::Immediate32),
                    _ => insn.set_op_kind(op_index, iced_x86::OpKind::Immediate64),
                }
                insn.set_immediate_i64(op_index, imm.as_i64());
            }
            IrOperand::Memory(mem) => {
                insn.set_op_kind(op_index, iced_x86::OpKind::Memory);
                insn.set_memory_base(mem.base.map(|r| self.convert_register(r)).unwrap_or(IcedRegister::None));
                insn.set_memory_index(mem.index.map(|r| self.convert_register(r)).unwrap_or(IcedRegister::None));
                insn.set_memory_index_scale(mem.scale);
                insn.set_memory_displacement64(mem.displacement as u64);
            }
        }
        Ok(())
    }

    fn convert_register(&self, reg: IrRegister) -> IcedRegister {
        match reg {
            IrRegister::Rax => IcedRegister::RAX,
            IrRegister::Rcx => IcedRegister::RCX,
            IrRegister::Rdx => IcedRegister::RDX,
            IrRegister::Rbx => IcedRegister::RBX,
            IrRegister::Rsp => IcedRegister::RSP,
            IrRegister::Rbp => IcedRegister::RBP,
            IrRegister::Rsi => IcedRegister::RSI,
            IrRegister::Rdi => IcedRegister::RDI,
            _ => IcedRegister::None,
        }
    }
}

pub struct EncodedInstruction {
    pub bytes: Vec<u8>,
    pub len: u8,
}

pub fn encode_single(instruction: &IrInstruction, mode: DisassemblyMode, base_address: u64) -> DisassemblyResult<EncodedInstruction> {
    let encoder = Encoder::new(mode);
    encoder.encode(instruction, base_address)
}

pub fn encode_all(instructions: &[IrInstruction], mode: DisassemblyMode, base_address: u64) -> DisassemblyResult<EncodedInstructions> {
    let encoder = Encoder::new(mode);
    let mut all_bytes = Vec::new();
    let mut offsets = Vec::new();
    let mut current_offset = 0;

    for insn in instructions {
        let encoded = encoder.encode(insn, base_address.wrapping_add(current_offset))?;
        offsets.push(current_offset);
        current_offset += encoded.len as u64;
        all_bytes.extend_from_slice(&encoded.bytes);
    }

    Ok(EncodedInstructions {
        bytes: all_bytes,
        offsets,
    })
}

pub struct EncodedInstructions {
    pub bytes: Vec<u8>,
    pub offsets: Vec<u64>,
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_encode_nop() {
        let insn = IrInstruction::new(0, IrOpcode::Nop);
        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        assert_eq!(result.bytes, vec![0x90]);
    }

    #[test]
    fn test_encode_push_rax() {
        let insn = IrInstruction::new(0, IrOpcode::Push {
            src: IrOperand::Register(IrRegister::Rax),
        });
        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        assert_eq!(result.bytes, vec![0x50]);
    }

    #[test]
    fn test_encode_pop_rax() {
        let insn = IrInstruction::new(0, IrOpcode::Pop {
            dst: IrOperand::Register(IrRegister::Rax),
        });
        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        assert_eq!(result.bytes, vec![0x58]);
    }

    #[test]
    fn test_encode_mov_rax_rbx() {
        let insn = IrInstruction::new(0, IrOpcode::Mov {
            dst: IrOperand::Register(IrRegister::Rax),
            src: IrOperand::Register(IrRegister::Rbx),
        });
        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        // mov rax, rbx is 48 8b c3 or 48 89 d8 depending on encoding
        assert!(result.bytes == vec![0x48, 0x89, 0xd8] || result.bytes == vec![0x48, 0x8b, 0xc3]);
    }

    #[test]
    fn test_encode_jmp_direct() {
        let insn = IrInstruction::new(0, IrOpcode::Jmp {
            target: IrJumpTarget::Direct(0x1005),
        });
        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        // Short jmp to $+5 from 0x1000 is EB 03 (since jmp is 2 bytes, 1002+3=1005)
        assert_eq!(result.bytes, vec![0xEB, 0x03]);
    }

    #[test]
    fn test_encode_all() {
        let instructions = vec![
            IrInstruction::new(0, IrOpcode::Push { src: IrOperand::Register(IrRegister::Rax) }),
            IrInstruction::new(1, IrOpcode::Pop { dst: IrOperand::Register(IrRegister::Rbx) }),
        ];
        let result = encode_all(&instructions, DisassemblyMode::Mode64, 0x1000).unwrap();
        assert_eq!(result.bytes, vec![0x50, 0x5b]);
        assert_eq!(result.offsets, vec![0, 1]);
    }

    #[test]
    fn test_complex_memory_addressing() {
        let mem = IrMemoryOperand {
            segment: None,
            base: Some(IrRegister::Rbx),
            index: Some(IrRegister::Rcx),
            scale: 4,
            displacement: 0,
            size_bits: 64,
        };

        let insn = IrInstruction::new(0, IrOpcode::Mov {
            dst: IrOperand::Register(IrRegister::Rax),
            src: IrOperand::Memory(mem),
        });

        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        assert!(result.bytes.len() >= 3);
        assert_eq!(result.bytes[0], 0x48);
        assert_eq!(result.bytes[1], 0x8B);
    }

    #[test]
    fn test_rip_relative_addressing() {
        let mem = IrMemoryOperand::new_rip_relative(0x100, 64);

        let insn = IrInstruction::new(0, IrOpcode::Mov {
            dst: IrOperand::Register(IrRegister::Rax),
            src: IrOperand::Memory(mem),
        });

        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        assert_eq!(result.bytes.len(), 7);
        assert_eq!(result.bytes[0], 0x48);
        assert_eq!(result.bytes[1], 0x8B);
        assert_eq!(result.bytes[2], 0x05);
    }

    #[test]
    fn test_memory_with_displacement() {
        let mem = IrMemoryOperand::new_base_disp(IrRegister::Rbx, 0x10, 64);

        let insn = IrInstruction::new(0, IrOpcode::Mov {
            dst: IrOperand::Register(IrRegister::Rax),
            src: IrOperand::Memory(mem),
        });

        let result = encode_single(&insn, DisassemblyMode::Mode64, 0x1000).unwrap();
        assert_eq!(result.bytes.len(), 4);
        assert_eq!(result.bytes[0], 0x48);
        assert_eq!(result.bytes[1], 0x8B);
        assert_eq!(result.bytes[2], 0x43);
        assert_eq!(result.bytes[3], 0x10);
    }
}
