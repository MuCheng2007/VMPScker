//! 虚拟机架构与上下文配置模块
//! - 固定寄存器分配
//! - 可插拔操作码映射 (OpcodeMapping trait)
//! - 特性开关 (FeatureFlags)
//! - 可选加密层 (CryptoConfig)
//! - 随机序言配置 (PrologueConfig)

use super::opcode::VmOpcode;
use iced_x86::code_asm::*;
use std::collections::HashMap;

// ========== 寄存器上下文 ==========

/// 虚拟机控制上下文 (固定寄存器分配)
/// r10=VIP, r14=VSP, r9=table, r11=scratch1, r8=scratch2, r12=vbase, rsi=image_base
#[derive(Debug, Clone, Copy)]
pub struct VmRegContext {
    pub vip: AsmRegister64,
    pub vsp: AsmRegister64,
    pub table: AsmRegister64,
    pub scratch1: AsmRegister64,
    pub scratch1_32: AsmRegister32,
    pub scratch2: AsmRegister64,
    pub vbase: AsmRegister64,
    pub image_base: AsmRegister64,
}

impl VmRegContext {
    /// VPushCtxReg 索引 → VM 上下文寄存器
    /// 0=vip, 1=vsp, 2=reserved, 3=table, 4=scratch1, 5=scratch2, 6=vbase, 7=image_base
    pub fn ctx_reg_by_index(&self, index: u8) -> AsmRegister64 {
        match index {
            0 => self.vip,
            1 => self.vsp,
            3 => self.table,
            4 => self.scratch1,
            5 => self.scratch2,
            6 => self.vbase,
            7 => self.image_base,
            _ => self.scratch1,
        }
    }
}

// ========== 操作码映射 trait ==========

/// 操作码 ↔ 字节码映射 trait
/// 后期可替换为 ShuffledMapping 实现随机混淆
pub trait OpcodeMapping {
    fn opcode_to_byte(&self, op: &VmOpcode) -> u8;
    fn opcode_from_byte(&self, byte: u8) -> VmOpcode;
    fn max_opcode_byte(&self) -> u8; // 最大有效 opcode 字节值（用于预填 handler table）
}

/// 明文 identity 映射（当前使用）
#[derive(Debug, Clone)]
pub struct IdentityMapping;

impl OpcodeMapping for IdentityMapping {
    fn opcode_to_byte(&self, op: &VmOpcode) -> u8 {
        match op {
            // 核心指令 (0-2)
            VmOpcode::VNop => 0,
            VmOpcode::VExit => 1,
            VmOpcode::VExec => 2,
            // 算术 (3-10)
            VmOpcode::VAdd => 3,
            VmOpcode::VSub => 4,
            VmOpcode::VXor => 5,
            VmOpcode::VNand => 6,
            VmOpcode::VNor => 7,
            VmOpcode::VMul => 8,
            VmOpcode::VDiv => 9,
            VmOpcode::VIdiv => 10,
            // 栈操作 (11)
            VmOpcode::VDrop => 11,
            // 控制流 (12-14)
            VmOpcode::VJmp(_) => 12,
            VmOpcode::VCall(_) => 13,
            // 立即数 (14-15)
            VmOpcode::VPushImm32(_) => 14,
            VmOpcode::VPushImm64(_) => 15,
            // 寄存器 (16-18)
            VmOpcode::VPushReg(_) => 16,
            VmOpcode::VPopReg(_) => 17,
            VmOpcode::VPushCtxReg(_) => 18,
            // 内存 (19-20)
            VmOpcode::VReadMem(_) => 19,
            VmOpcode::VWriteMem(_) => 20,
            // 标签 (21)
            VmOpcode::VLabel(_) => 21,
            // 条件跳转 (22-37)
            VmOpcode::VJcc(cond, _) => 22 + cond,
            _ => 0,
        }
    }

    fn opcode_from_byte(&self, byte: u8) -> VmOpcode {
        match byte {
            0 => VmOpcode::VNop,
            1 => VmOpcode::VExit,
            2 => VmOpcode::VExec,
            3 => VmOpcode::VAdd,
            4 => VmOpcode::VSub,
            5 => VmOpcode::VXor,
            6 => VmOpcode::VNand,
            7 => VmOpcode::VNor,
            8 => VmOpcode::VMul,
            9 => VmOpcode::VDiv,
            10 => VmOpcode::VIdiv,
            11 => VmOpcode::VDrop,
            12 => VmOpcode::VJmp(0),
            13 => VmOpcode::VCall(0),
            14 => VmOpcode::VPushImm32(0),
            15 => VmOpcode::VPushImm64(0),
            16 => VmOpcode::VPushReg(0),
            17 => VmOpcode::VPopReg(0),
            18 => VmOpcode::VPushCtxReg(0),
            19 => VmOpcode::VReadMem(0),
            20 => VmOpcode::VWriteMem(0),
            21 => VmOpcode::VLabel(0),
            22..=37 => VmOpcode::VJcc(byte - 22, 0),
            _ => VmOpcode::VNop,
        }
    }

    fn max_opcode_byte(&self) -> u8 {
        37
    }
}

// ========== 特性开关 ==========

/// 虚拟机特性开关
/// 所有开关初始为 false，后期可逐步开启
#[derive(Debug, Clone)]
pub struct FeatureFlags {
    pub encrypt_bytecode: bool,  // 滚动加密开关
    pub random_prologue: bool,   // 随机序言/尾声
    pub handler_fusion: bool,    // handler 融合
}

impl Default for FeatureFlags {
    fn default() -> Self {
        Self {
            encrypt_bytecode: false,
            random_prologue: false,
            handler_fusion: false,
        }
    }
}

// ========== 加密配置 (预留) ==========

/// 加密操作类型
#[derive(Debug, Clone, Copy)]
pub enum CryptoOp {
    Xor,
    Add,
    Sub,
}

/// 字节码加密配置
/// 当 FeatureFlags::encrypt_bytecode 为 true 时启用
#[derive(Debug, Clone)]
pub struct CryptoConfig {
    pub initial_key: u32,
    pub chain: Vec<CryptoOp>,
}

impl CryptoConfig {
    pub fn new_random() -> Self {
        use rand::Rng;
        let mut rng = rand::thread_rng();
        let chain_len = rng.gen_range(2..=5);
        let chain: Vec<CryptoOp> = (0..chain_len)
            .map(|_| match rng.gen_range(0..3) {
                0 => CryptoOp::Xor,
                1 => CryptoOp::Add,
                _ => CryptoOp::Sub,
            })
            .collect();
        Self {
            initial_key: rng.gen(),
            chain,
        }
    }

    pub fn encrypt(&self, plain: u32, key: u32) -> u32 {
        let mut val = plain;
        for op in &self.chain {
            match op {
                CryptoOp::Xor => val ^= key,
                CryptoOp::Add => val = val.wrapping_add(key),
                CryptoOp::Sub => val = val.wrapping_sub(key),
            }
        }
        val
    }

    pub fn decrypt(&self, cipher: u32, key: u32) -> u32 {
        let mut val = cipher;
        for op in self.chain.iter().rev() {
            match op {
                CryptoOp::Xor => val ^= key,
                CryptoOp::Add => val = val.wrapping_sub(key),
                CryptoOp::Sub => val = val.wrapping_add(key),
            }
        }
        val
    }
}

// ========== 序言配置 (预留) ==========

/// 随机序言配置
/// 当 FeatureFlags::random_prologue 为 true 时启用
#[derive(Debug, Clone)]
pub struct PrologueConfig {
    /// VM 上下文寄存器压栈顺序（索引 0-7 的排列）
    pub reg_order: Vec<u8>,
}

impl Default for PrologueConfig {
    fn default() -> Self {
        Self {
            reg_order: vec![0, 1, 2, 3, 4, 5, 6, 7],
        }
    }
}

impl PrologueConfig {
    pub fn new_random() -> Self {
        use rand::seq::SliceRandom;
        use rand::thread_rng;
        let mut order: Vec<u8> = (0..8).collect();
        order.shuffle(&mut thread_rng());
        Self { reg_order: order }
    }
}

// ========== ArchConfig ==========

/// 全局架构配置
#[derive(Debug, Clone)]
pub struct ArchConfig {
    pub context: VmRegContext,
    pub mapping: IdentityMapping,
    pub features: FeatureFlags,
    pub crypto: Option<CryptoConfig>,
    pub prologue: PrologueConfig,
}

impl ArchConfig {
    /// 创建默认架构配置（全特性关闭）
    /// r10=VIP, r14=VSP, r9=table, r11=scratch1, r8=scratch2, r12=vbase, rsi=image_base
    pub fn new_default() -> Self {
        let context = VmRegContext {
            vip: r10,
            vsp: r14,
            table: r9,
            scratch1: r11,
            scratch1_32: r11d,
            scratch2: r8,
            vbase: r12,
            image_base: rsi,
        };

        Self {
            context,
            mapping: IdentityMapping,
            features: FeatureFlags::default(),
            crypto: None,
            prologue: PrologueConfig::default(),
        }
    }
}

// ========== 便捷全局函数（使用 IdentityMapping） ==========

/// 使用 IdentityMapping 的 opcode → byte 转换
pub fn opcode_to_byte(op: &VmOpcode) -> u8 {
    IdentityMapping.opcode_to_byte(op)
}

/// 使用 IdentityMapping 的 byte → opcode 转换
pub fn opcode_from_byte(byte: u8) -> VmOpcode {
    IdentityMapping.opcode_from_byte(byte)
}
