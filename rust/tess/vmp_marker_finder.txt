//! VMP 标记查找器
//!
//! 通过分析 PE 文件的导入表和调用指令来定位 VMP 标记
//! VMProtect 标记是通过调用 SDK DLL 函数实现的

use crate::error::{Result, VmpError};
use crate::pe::file::PeFile;
use crate::intel::{DisassemblyMode, disassemble};

/// VMP SDK 函数名称
const VMP_BEGIN_FUNCTIONS: &[&str] = &[
    "VMProtectBegin",
    "VMProtectBeginVirtualization",
    "VMProtectBeginMutation",
    "VMProtectBeginUltra",
    "VMProtectBeginVirtualizationLockByKey",
    "VMProtectBeginUltraLockByKey",
];

const VMP_END_FUNCTION: &str = "VMProtectEnd";

/// 标记类型
#[derive(Debug, Clone, Copy, PartialEq, Eq)]
pub enum VmpMarkerType {
    /// 基本保护
    Begin,
    /// 虚拟化保护
    Virtualization,
    /// 变异保护
    Mutation,
    /// Ultra 保护
    Ultra,
    /// 虚拟化 + 密钥锁定
    VirtualizationLockByKey,
    /// Ultra + 密钥锁定
    UltraLockByKey,
    /// 结束标记
    End,
}

impl VmpMarkerType {
    /// 从函数名解析标记类型
    pub fn from_function_name(name: &str) -> Option<Self> {
        match name {
            "VMProtectBegin" => Some(VmpMarkerType::Begin),
            "VMProtectBeginVirtualization" => Some(VmpMarkerType::Virtualization),
            "VMProtectBeginMutation" => Some(VmpMarkerType::Mutation),
            "VMProtectBeginUltra" => Some(VmpMarkerType::Ultra),
            "VMProtectBeginVirtualizationLockByKey" => Some(VmpMarkerType::VirtualizationLockByKey),
            "VMProtectBeginUltraLockByKey" => Some(VmpMarkerType::UltraLockByKey),
            "VMProtectEnd" => Some(VmpMarkerType::End),
            _ => None,
        }
    }
}

impl std::fmt::Display for VmpMarkerType {
    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
        match self {
            VmpMarkerType::Begin => write!(f, "Begin"),
            VmpMarkerType::Virtualization => write!(f, "Virtualization"),
            VmpMarkerType::Mutation => write!(f, "Mutation"),
            VmpMarkerType::Ultra => write!(f, "Ultra"),
            VmpMarkerType::VirtualizationLockByKey => write!(f, "VirtualizationLockByKey"),
            VmpMarkerType::UltraLockByKey => write!(f, "UltraLockByKey"),
            VmpMarkerType::End => write!(f, "End"),
        }
    }
}

/// VMP 标记位置
#[derive(Debug, Clone)]
pub struct VmpMarkerLocation {
    /// 调用指令的 RVA
    pub call_rva: u64,
    /// 调用指令的文件偏移
    pub call_file_offset: usize,
    /// 目标函数 RVA（导入表 thunk）
    pub target_rva: u64,
    /// 标记类型
    pub marker_type: VmpMarkerType,
    /// 函数名
    pub function_name: String,
}

/// VMP 标记对
#[derive(Debug, Clone)]
pub struct VmpMarkerPair {
    /// Begin 标记
    pub begin: VmpMarkerLocation,
    /// End 标记
    pub end: VmpMarkerLocation,
    /// 保护的代码起始 RVA（Begin 调用后的下一条指令）
    pub code_start: u64,
    /// 保护的代码结束 RVA（End 调用的地址）
    pub code_end: u64,
}

/// VMP 标记查找器
pub struct VmpMarkerFinder {
    mode: DisassemblyMode,
}

impl VmpMarkerFinder {
    /// 创建新的查找器
    pub fn new() -> Self {
        Self {
            mode: DisassemblyMode::Mode64,
        }
    }

    /// 创建指定模式的查找器
    pub fn with_mode(mode: DisassemblyMode) -> Self {
        Self { mode }
    }

    /// 查找 PE 文件中的所有 VMP 标记
    pub fn find_markers(&self, pe_file: &PeFile) -> Result<Vec<VmpMarkerLocation>> {
        let mut markers = Vec::new();

        // 获取导入表信息 (IAT_RVA, Function_Name)
        let imports = self.get_vmp_imports(pe_file)?;
        eprintln!("DEBUG: Found {} VMP imports", imports.len());
        for (iat_rva, name) in &imports {
            eprintln!("DEBUG:   {} @ IAT={:08X}", name, iat_rva);
        }
        
        if imports.is_empty() {
            return Ok(markers);
        }

        // 扫描代码节查找对这些导入的调用
        let code_sections = self.get_code_sections(pe_file)?;
        eprintln!("DEBUG: Found {} code sections", code_sections.len());
        
        for (section_rva, section_data) in code_sections {
            eprintln!("DEBUG: Scanning section at RVA {:08X}, size {}", section_rva, section_data.len());
            let calls = self.find_call_instructions(&section_data, section_rva)?;
            eprintln!("DEBUG: Found {} call instructions", calls.len());
            
            for (call_rva, target_rva) in calls {
                eprintln!("DEBUG: Call at {:08X} -> IAT {:08X}", call_rva, target_rva);
                // 检查目标是否是 VMP 函数
                if let Some((marker_type, func_name)) = self.is_vmp_import(target_rva, &imports) {
                    eprintln!("DEBUG:   -> MATCH: {}", func_name);
                    let file_offset = self.rva_to_file_offset(pe_file, call_rva)?;
                    
                    markers.push(VmpMarkerLocation {
                        call_rva,
                        call_file_offset: file_offset,
                        target_rva,
                        marker_type,
                        function_name: func_name.to_string(),
                    });
                }
            }
        }

        // 按地址排序
        markers.sort_by_key(|m| m.call_rva);

        Ok(markers)
    }

    /// 查找标记对
    pub fn find_marker_pairs(&self, pe_file: &PeFile) -> Result<Vec<VmpMarkerPair>> {
        let markers = self.find_markers(pe_file)?;
        let mut pairs = Vec::new();
        let mut stack: Vec<VmpMarkerLocation> = Vec::new();

        for marker in markers {
            match marker.marker_type {
                VmpMarkerType::End => {
                    if let Some(begin) = stack.pop() {
                        // 计算代码范围
                        let code_start = begin.call_rva + self.get_call_instruction_size(pe_file, begin.call_rva)?;
                        let code_end = marker.call_rva;

                        pairs.push(VmpMarkerPair {
                            begin,
                            end: marker,
                            code_start,
                            code_end,
                        });
                    }
                }
                _ => {
                    stack.push(marker);
                }
            }
        }

        Ok(pairs)
    }

    /// 查找虚拟化标记对并显示汇编代码
    pub fn find_virtualization_markers(&self, pe_file: &PeFile) -> Result<Vec<(VmpMarkerPair, String)>> {
        let pairs = self.find_marker_pairs(pe_file)?;
        let mut result = Vec::new();

        for pair in pairs {
            // 只处理 Virtualization 类型的标记
            if pair.begin.marker_type == VmpMarkerType::Virtualization {
                let asm = self.disassemble_protected_code(pe_file, &pair)?;
                result.push((pair, asm));
            }
        }

        Ok(result)
    }

    /// 获取 VMP SDK 导入函数
    /// 返回 (IAT_RVA, Function_Name) 的列表
    fn get_vmp_imports(&self, pe_file: &PeFile) -> Result<Vec<(u64, String)>> {
        let mut imports = Vec::new();

        if let Some(pe) = pe_file.pe() {
            // 使用 goblin 解析导入表
            use goblin::pe::import::ImportData;
            
            // 获取可选头中的数据目录
            if let Some(optional_header) = pe.header.optional_header {
                let data_dirs = &optional_header.data_directories;
                
                // 导入表是第2个数据目录（索引1）
                let import_dir_index = 1; // IMAGE_DIRECTORY_ENTRY_IMPORT
                if let Some(Some((_, import_dir))) = data_dirs.data_directories.get(import_dir_index) {
                    
                    if import_dir.virtual_address > 0 && import_dir.size > 0 {
                        let file_data = pe_file.data();
                        
                        // 解析导入表
                        let file_alignment = optional_header.windows_fields.file_alignment;
                        let is_64bit = pe.header.coff_header.machine == goblin::pe::header::COFF_MACHINE_X86_64;
                        let import_result = if is_64bit {
                            ImportData::parse::<u64>(file_data, *import_dir, &pe.sections, file_alignment)
                        } else {
                            ImportData::parse::<u32>(file_data, *import_dir, &pe.sections, file_alignment)
                        };
                        
                        if let Ok(import_data) = import_result {
                            for entry in &import_data.import_data {
                                let dll_name = entry.name.to_string();
                                
                                // 检查是否是 VMProtect SDK DLL
                                if dll_name.to_lowercase().contains("vmprotect") {
                                    // 获取 IAT RVA
                                    let iat_rva = entry.import_directory_entry.import_address_table_rva;
                                    
                                    // 遍历导入函数
                                    if let Some(ref lookup_table) = entry.import_lookup_table {
                                        for (i, lookup_entry) in lookup_table.iter().enumerate() {
                                            match lookup_entry {
                                                goblin::pe::import::SyntheticImportLookupTableEntry::HintNameTableRVA((_, hint_entry)) => {
                                                    let func_name = hint_entry.name.to_string();
                                                    if VMP_BEGIN_FUNCTIONS.contains(&func_name.as_str()) || 
                                                       func_name == VMP_END_FUNCTION {
                                                        // 计算 IAT 条目 RVA (64-bit = 8 bytes per entry)
                                                        let iat_entry_rva = iat_rva as u64 + (i * 8) as u64;
                                                        imports.push((iat_entry_rva, func_name));
                                                    }
                                                }
                                                _ => {}
                                            }
                                        }
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }

        Ok(imports)
    }

    /// 获取代码节
    fn get_code_sections(&self, pe_file: &PeFile) -> Result<Vec<(u64, Vec<u8>)>> {
        let mut sections = Vec::new();

        if let Some(pe) = pe_file.pe() {
            for section in &pe.sections {
                let name = String::from_utf8_lossy(&section.name)
                    .trim_end_matches('\0')
                    .to_string();
                
                // 检查是否是代码节
                if name == ".text" || name == ".code" || 
                   (section.characteristics & 0x20000000) != 0 { // IMAGE_SCN_CNT_CODE
                    let rva = section.virtual_address as u64;
                    let size = section.size_of_raw_data as usize;
                    let offset = section.pointer_to_raw_data as usize;
                    
                    let pe_data = pe_file.data();
                    if offset + size <= pe_data.len() {
                        let data = pe_data[offset..offset + size].to_vec();
                        sections.push((rva, data));
                    }
                }
            }
        }

        Ok(sections)
    }

    /// 查找 call 指令
    fn find_call_instructions(&self, code: &[u8], base_rva: u64) -> Result<Vec<(u64, u64)>> {
        let mut calls = Vec::new();
        
        // 反汇编代码
        let instructions = disassemble(code, self.mode, base_rva)
            .map_err(|e| VmpError::vm_error(format!("Disassembly failed: {:?}", e)))?;

        for inst in instructions {
            let mnemonic = inst.mnemonic();
            
            // 检查是否是 call 指令
            if mnemonic == "call" {
                // 尝试获取调用目标
                if let Some(target) = self.extract_call_target(&inst) {
                    calls.push((inst.ip(), target));
                }
            }
        }

        Ok(calls)
    }

    /// 从 call 指令中提取目标地址
    fn extract_call_target(&self, inst: &crate::intel::Instruction) -> Option<u64> {
        // 获取 iced_x86 指令
        let iced = inst.iced();
        
        // 检查是否是相对调用 (E8 xx xx xx xx) 或绝对调用
        use iced_x86::OpKind;
        
        match iced.op0_kind() {
            OpKind::NearBranch64 | OpKind::NearBranch32 | OpKind::NearBranch16 => {
                Some(iced.near_branch_target())
            }
            OpKind::Memory => {
                // 间接调用 [address] - 例如 call qword ptr [rip+0x...]
                // 尝试从内存操作数中提取地址
                self.extract_memory_address(&iced)
            }
            _ => None,
        }
    }

    /// 从内存操作数中提取地址
    fn extract_memory_address(&self, iced: &iced_x86::Instruction) -> Option<u64> {
        use iced_x86::{OpKind, Register};
        
        // 检查是否是内存操作数
        if iced.op0_kind() != OpKind::Memory {
            return None;
        }
        
        // 获取内存操作数的基址寄存器、索引寄存器和位移
        let base = iced.memory_base();
        let index = iced.memory_index();
        
        // 对于 RIP 相对寻址（x64 常见），base = RIP
        if base == Register::RIP {
            // iced_x86 的 memory_displacement32() 对于 RIP 相对寻址
            // 返回的是绝对地址（已经计算好了），不是相对位移！
            let target = iced.memory_displacement64();
            eprintln!("DEBUG: RIP-relative: ip={:08X}, target={:08X}", iced.ip(), target);
            return Some(target);
        }
        
        // 对于绝对地址 [displ]
        if base == Register::None && index == Register::None {
            return Some(iced.memory_displacement64());
        }
        
        // 其他情况（基址+索引+位移）无法静态确定
        None
    }

    /// 检查目标地址是否是 VMP 导入
    /// 注意：target_rva 是 call 指令的目标，应该是 IAT 地址
    fn is_vmp_import<'a>(&self, target_rva: u64, imports: &'a [(u64, String)]) -> Option<(VmpMarkerType, &'a str)> {
        // 直接匹配 IAT 地址
        for (iat_rva, func_name) in imports {
            if *iat_rva == target_rva {
                if let Some(marker_type) = VmpMarkerType::from_function_name(func_name) {
                    eprintln!("DEBUG: is_vmp_import MATCH: {} @ {:08X}", func_name, target_rva);
                    return Some((marker_type, func_name));
                }
            }
        }
        
        None
    }

    /// RVA 转换为文件偏移
    fn rva_to_file_offset(&self, pe_file: &PeFile, rva: u64) -> Result<usize> {
        if let Some(pe) = pe_file.pe() {
            for section in &pe.sections {
                let section_rva = section.virtual_address as u64;
                let section_size = section.virtual_size as u64;

                if rva >= section_rva && rva < section_rva + section_size {
                    let offset = section.pointer_to_raw_data as usize + (rva - section_rva) as usize;
                    return Ok(offset);
                }
            }
        }
        Err(VmpError::vm_error(format!("RVA 0x{:X} not found in any section", rva)))
    }

    /// 获取 call 指令的大小
    pub fn get_call_instruction_size(&self, pe_file: &PeFile, call_rva: u64) -> Result<u64> {
        let file_offset = self.rva_to_file_offset(pe_file, call_rva)?;
        let pe_data = pe_file.data();
        
        // 读取足够的字节进行反汇编
        let end = std::cmp::min(file_offset + 15, pe_data.len());
        let code = &pe_data[file_offset..end];
        
        let instructions = disassemble(code, self.mode, call_rva)
            .map_err(|e| VmpError::vm_error(format!("Disassembly failed: {:?}", e)))?;
        
        if let Some(first) = instructions.first() {
            Ok(first.len() as u64)
        } else {
            // 默认 call 指令大小
            Ok(5)
        }
    }

    /// 读取标记之间的代码
    pub fn read_protected_code(&self, pe_file: &PeFile, pair: &VmpMarkerPair) -> Result<Vec<u8>> {
        let start_offset = self.rva_to_file_offset(pe_file, pair.code_start)?;
        let end_offset = self.rva_to_file_offset(pe_file, pair.code_end)?;

        if end_offset <= start_offset {
            return Ok(Vec::new());
        }

        let pe_data = pe_file.data();
        if end_offset > pe_data.len() {
            return Err(VmpError::vm_error("Invalid code range".to_string()));
        }

        Ok(pe_data[start_offset..end_offset].to_vec())
    }

    /// 反汇编保护的代码
    pub fn disassemble_protected_code(&self, pe_file: &PeFile, pair: &VmpMarkerPair) -> Result<String> {
        let code_bytes = self.read_protected_code(pe_file, pair)?;
        if code_bytes.is_empty() {
            return Ok("// No code between markers".to_string());
        }

        let instructions = disassemble(&code_bytes, self.mode, pair.code_start)
            .map_err(|e| VmpError::vm_error(format!("Failed to disassemble: {:?}", e)))?;

        let mut output = String::new();
        for inst in instructions {
            output.push_str(&format!("0x{:08X}: {}\n", inst.ip(), inst.mnemonic()));
        }

        Ok(output)
    }
}

impl Default for VmpMarkerFinder {
    fn default() -> Self {
        Self::new()
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_marker_type_from_function_name() {
        assert_eq!(
            VmpMarkerType::from_function_name("VMProtectBegin"),
            Some(VmpMarkerType::Begin)
        );
        assert_eq!(
            VmpMarkerType::from_function_name("VMProtectBeginVirtualization"),
            Some(VmpMarkerType::Virtualization)
        );
        assert_eq!(
            VmpMarkerType::from_function_name("VMProtectEnd"),
            Some(VmpMarkerType::End)
        );
        assert_eq!(
            VmpMarkerType::from_function_name("SomeOtherFunction"),
            None
        );
    }

    #[test]
    fn test_iced_displacement() {
        // 测试 iced_x86 是否正确解析 RIP 相对寻址
        // 指令: call qword ptr [rip+0x30FE] at RVA 0x1024
        // 机器码: FF 15 FE 30 00 00
        let data = [0xFF, 0x15, 0xFE, 0x30, 0x00, 0x00];
        let ip = 0x1024u64;
        
        let mut decoder = iced_x86::Decoder::with_ip(64, &data, ip, iced_x86::DecoderOptions::NONE);
        
        let insn = decoder.decode();
        println!("Instruction: {}", insn);
        println!("IP: 0x{:08X}", insn.ip());
        println!("Next IP: 0x{:08X}", insn.next_ip());
        println!("Memory displacement32: 0x{:08X}", insn.memory_displacement32());
        println!("Memory displacement64: 0x{:016X}", insn.memory_displacement64());
        
        // iced_x86 的 memory_displacement 对于 RIP 相对寻址返回的是绝对地址！
        let target = insn.memory_displacement64();
        println!("Target from memory_displacement64: 0x{:08X}", target);
        println!("Expected target: 0x4128");
        
        // 验证
        assert_eq!(target, 0x4128, "Target address should be 0x4128");
    }
}
